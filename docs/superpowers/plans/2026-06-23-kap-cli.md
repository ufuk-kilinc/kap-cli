# KAP CLI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A Python CLI (`kap`) that wraps all 11 KAP Veri Yayın Servisi endpoints as easy-to-use commands for the research team.

**Architecture:** Thin layered client — `config` (typed `.env` settings) → `auth` (JWT mint + disk cache) → `client` (one method per endpoint, retry-once on token error) → `render` (table/json/csv) → `cli` (Typer commands). Faithful wrapper over the C# `TuneUp.Integrations.Kap` contract; no business logic.

**Tech Stack:** Python 3.11+, Typer (CLI), Rich (tables), httpx (HTTP), pydantic-settings (config), pytest + respx (tests).

## Global Constraints

- Python **3.11+**.
- Token sent in `Authorization` header **RAW — no `Bearer` prefix** (Bearer → ER012).
- All service paths under `/api/vyk/`; `generateToken` under `/auth/`.
- KAP errors are **non-2xx** responses with a JSON envelope `{"code":"ERxxx","message":"..."}`.
  - `ER005` → "not found" → map to empty list / `None`.
  - `ER004` / `ER006` / `ER012` → token error → invalidate, re-mint, retry **once**.
  - other `ERxxx` → raise `KapApiError`.
- `downloadAttachment` body is a Java-serialized envelope; strip to the `%PDF` marker.
- API keys are secrets → `.env` is gitignored; only `.env.example` is committed.
- Hosts: test `https://vykapipreprod.mkk.com.tr`, prod `https://kapdatafeed.kap.org.tr`.
- DRY, YAGNI, TDD, frequent commits.

```python
from __future__ import annotations

from dataclasses import dataclass

from pydantic_settings import BaseSettings, SettingsConfigDict


@dataclass(frozen=True)
class ResolvedConfig:
    env: str
    base_url: str
    api_key: str
    token_refresh_skew_seconds: int


class Settings(BaseSettings):
    """KAP CLI configuration, loaded from environment / .env (prefix KAP_)."""

    model_config = SettingsConfigDict(
        env_prefix="KAP_",
        env_file=".env",
        env_file_encoding="utf-8",
        extra="ignore",
    )

    active_env: str = "test"
    test_base_url: str = "https://vykapipreprod.mkk.com.tr"
    test_api_key: str = ""
    prod_base_url: str = "https://kapdatafeed.kap.org.tr"
    prod_api_key: str = ""
    token_refresh_skew_seconds: int = 60

    def resolve(self, env_override: str | None = None) -> ResolvedConfig:
        env = (env_override or self.active_env).strip().lower()
        if env == "test":
            name, base_url, api_key, key_var = (
                "test", self.test_base_url, self.test_api_key, "KAP_TEST_API_KEY",
            )
        elif env in ("production", "prod"):
            name, base_url, api_key, key_var = (
                "production", self.prod_base_url, self.prod_api_key, "KAP_PROD_API_KEY",
            )
        else:
            raise ValueError(f"Unknown env '{env}' (use 'test' or 'production').")

        if not api_key:
            raise ValueError(
                f"Missing API key for env '{name}'. Set {key_var} in your .env file."
            )

        return ResolvedConfig(
            env=name,
            base_url=base_url.rstrip("/"),
            api_key=api_key,
            token_refresh_skew_seconds=max(0, self.token_refresh_skew_seconds),
        )


def load_config(env_override: str | None = None) -> ResolvedConfig:
    return Settings().resolve(env_override)
```

```python
from __future__ import annotations

import json

_TOKEN_ERROR_CODES = {"ER004", "ER006", "ER012"}
_NOT_FOUND_CODE = "ER005"


class KapError(Exception):
    """Base class for all KAP CLI errors."""


class KapApiError(KapError):
    """A non-2xx response carrying a KAP error envelope {code, message}."""

    def __init__(self, status_code: int, code: str | None, message: str):
        self.status_code = status_code
        self.code = code
        self.message = message
        super().__init__(f"KAP {code or '?'} (HTTP {status_code}): {message}")

    @property
    def is_token_error(self) -> bool:
        return self.code in _TOKEN_ERROR_CODES

    @property
    def is_not_found(self) -> bool:
        return self.code == _NOT_FOUND_CODE

    @classmethod
    def from_body(cls, status_code: int, body: str) -> "KapApiError":
        code: str | None = None
        message = body
        try:
            parsed = json.loads(body)
            if isinstance(parsed, dict):
                code = parsed.get("code")
                message = parsed.get("message") or body
        except (ValueError, TypeError):
            pass  # not JSON (HTML error page etc.) — keep the raw body
        return cls(status_code, code, message)


class KapConnectivityError(KapError):
    """Network-level failure reaching KAP (often an IP-allowlist / VPN problem)."""
```

```python
from __future__ import annotations

import base64
import json
import time
from pathlib import Path

import httpx

from kap.errors import KapApiError


def parse_jwt_expiry(jwt: str) -> int | None:
    """Return the `exp` claim (unix seconds) from a JWT, without verifying its signature."""
    try:
        parts = jwt.split(".")
        if len(parts) < 2:
            return None
        payload = parts[1]
        payload += "=" * (-len(payload) % 4)  # restore base64 padding
        data = json.loads(base64.urlsafe_b64decode(payload))
        exp = data.get("exp")
        return int(exp) if exp is not None else None
    except (ValueError, TypeError, json.JSONDecodeError):
        return None


class TokenProvider:
    """Mints the KAP JWT and caches it on disk so each CLI process reuses it."""

    def __init__(
        self,
        base_url: str,
        api_key: str,
        env: str,
        skew_seconds: int = 60,
        cache_dir: Path | None = None,
        http: httpx.Client | None = None,
    ):
        self._base_url = base_url.rstrip("/")
        self._api_key = api_key
        self._env = env
        self._skew = max(0, skew_seconds)
        self._cache_dir = cache_dir or (Path.home() / ".kap")
        self._cache_path = self._cache_dir / f"token-{env}.json"
        self._http = http or httpx.Client(timeout=30.0)

    def get_token(self) -> str:
        cached = self._read_cache()
        if cached and not self._is_expiring(cached["expires_at"]):
            return cached["token"]
        return self._mint()

    def invalidate(self) -> None:
        try:
            self._cache_path.unlink(missing_ok=True)
        except OSError:
            pass

    # ── internals ────────────────────────────────────────────────────────────
    def _is_expiring(self, expires_at: int) -> bool:
        return time.time() >= (expires_at - self._skew)

    def _mint(self) -> str:
        resp = self._http.get(
            f"{self._base_url}/auth/generateToken", params={"apiKey": self._api_key}
        )
        if not resp.is_success:
            raise KapApiError.from_body(resp.status_code, resp.text)
        token = (resp.json() or {}).get("token")
        if not token:
            raise KapApiError(resp.status_code, None, f"generateToken returned no token: {resp.text}")
        # Fail safe on an unparseable exp: assume 1h so we re-mint soon.
        exp = parse_jwt_expiry(token) or int(time.time()) + 3600
        self._write_cache(token, exp)
        return token

    def _read_cache(self) -> dict | None:
        try:
            data = json.loads(self._cache_path.read_text())
            if isinstance(data, dict) and "token" in data and "expires_at" in data:
                return data
        except (OSError, ValueError):
            pass
        return None

    def _write_cache(self, token: str, expires_at: int) -> None:
        try:
            self._cache_dir.mkdir(parents=True, exist_ok=True)
            self._cache_path.write_text(json.dumps({"token": token, "expires_at": expires_at}))
            try:
                self._cache_path.chmod(0o600)
            except OSError:
                pass  # best-effort on platforms without POSIX perms
        except OSError:
            pass  # cache is an optimization; never fail the call over it
```

```python
from __future__ import annotations

import httpx

from kap.auth import TokenProvider
from kap.config import ResolvedConfig
from kap.errors import KapApiError, KapConnectivityError

_PDF_MARKER = b"%PDF"

_CONNECTIVITY_HINT = (
    "Could not connect to KAP. You may not be on an allowlisted IP — MKK silently "
    "drops non-allowlisted source IPs (TCP timeout, not 401). Check your VPN / egress "
    "IP, or ask MKK (kapdestek@mkk.com.tr) to allowlist it."
)


class KapClient:
    """Typed-ish client over the KAP /api/vyk/* endpoints (returns parsed JSON)."""

    def __init__(
        self,
        config: ResolvedConfig,
        token_provider: TokenProvider | None = None,
        http: httpx.Client | None = None,
    ):
        self._config = config
        self._auth = token_provider or TokenProvider(
            config.base_url, config.api_key, config.env, config.token_refresh_skew_seconds
        )
        self._http = http or httpx.Client(base_url=config.base_url, timeout=60.0)

    # ── public endpoints ──────────────────────────────────────────────────────
    def last_disclosure_index(self) -> int:
        data = self._get("api/vyk/lastDisclosureIndex").json() or {}
        for key, value in data.items():
            if key.lower() == "lastdisclosureindex":
                return int(value)
        raise KapApiError(200, None, f"Unexpected lastDisclosureIndex response: {data}")

    def disclosures(
        self,
        index: int,
        disclosure_type: str | None = None,
        disclosure_class: str | None = None,
        company_id: int | None = None,
    ) -> list:
        params: dict[str, str | int] = {"disclosureIndex": index}
        if disclosure_type:
            params["disclosureType"] = disclosure_type
        if disclosure_class:
            params["disclosureClass"] = disclosure_class
        if company_id is not None:
            params["companyId"] = company_id
        return self._get_list("api/vyk/disclosures", params)

    def disclosure_detail(
        self, index: int, file_type: str = "data", sub_report: str | None = None
    ) -> dict | None:
        params: dict[str, str] = {"fileType": file_type}
        if sub_report:
            params["subReportList"] = sub_report
        try:
            return self._get(f"api/vyk/disclosureDetail/{index}", params).json()
        except KapApiError as exc:
            if exc.is_not_found:
                return None
            raise

    def download_attachment(self, attachment_id: str) -> bytes | None:
        if not attachment_id or not attachment_id.strip():
            raise ValueError("attachment_id is required.")
        try:
            raw = self._get(f"api/vyk/downloadAttachment/{attachment_id}").content
        except KapApiError as exc:
            if exc.is_not_found:
                return None
            raise
        return self._extract_pdf(raw)

    def members(self) -> list:
        return self._get_list("api/vyk/members")

    def funds(self) -> list:
        return self._get_list("api/vyk/funds")

    def fund_detail(self, fund_id: int) -> list:
        return self._get_list(f"api/vyk/fundDetail/{fund_id}")

    def member_securities(self) -> list:
        return self._get_list("api/vyk/memberSecurities")

    def blocked_disclosures(self) -> list:
        return self._get_list("api/vyk/blockedDisclosures")

    def ca_event_status(self, ref_ids: list[int]) -> list:
        if not ref_ids:
            raise ValueError("At least one processRefId is required.")
        joined = ",".join(str(r) for r in ref_ids)
        return self._get_list("api/vyk/caEventStatus", {"processRefId": joined})

    def raw(self, path: str, query: dict[str, str] | None = None) -> dict:
        resp = self._send(path.lstrip("/"), query)
        return {
            "service": path,
            "status_code": resp.status_code,
            "is_success": resp.is_success,
            "content_type": resp.headers.get("content-type"),
            "body": resp.text,
        }

    def close(self) -> None:
        self._http.close()

    def __enter__(self) -> "KapClient":
        return self

    def __exit__(self, *exc) -> None:
        self.close()

    # ── internals ────────────────────────────────────────────────────────────
    def _get(self, path: str, params: dict | None = None) -> httpx.Response:
        """GET with raw-token auth + retry-once on token error; raises on non-2xx."""
        for attempt in range(2):
            resp = self._send(path, params)
            if resp.is_success:
                return resp
            err = KapApiError.from_body(resp.status_code, resp.text)
            if err.is_token_error and attempt == 0:
                self._auth.invalidate()
                continue
            raise err
        raise err  # pragma: no cover

    def _send(self, path: str, params: dict | None) -> httpx.Response:
        token = self._auth.get_token()
        try:
            return self._http.get(path, params=params, headers={"Authorization": token})
        except (httpx.ConnectError, httpx.ConnectTimeout) as exc:
            raise KapConnectivityError(_CONNECTIVITY_HINT) from exc

    def _get_list(self, path: str, params: dict | None = None) -> list:
        try:
            return self._get(path, params).json() or []
        except KapApiError as exc:
            if exc.is_not_found:
                return []
            raise

    @staticmethod
    def _extract_pdf(raw: bytes) -> bytes:
        if not raw:
            return raw
        start = raw.find(_PDF_MARKER)
        return raw[start:] if start > 0 else raw
```

```python
from __future__ import annotations

import base64
import csv
import io
import json
import re
from pathlib import Path

from rich.console import Console
from rich.table import Table

_console = Console()
_STYLE_SCRIPT = re.compile(r"<(style|script)\b[^>]*>.*?</\1>", re.IGNORECASE | re.DOTALL)
_TAGS = re.compile(r"<[^>]+>")
_WS = re.compile(r"\s+")


def _is_scalar(data) -> bool:
    return not isinstance(data, (dict, list))


def resolve_format(fmt: str | None, out: Path | None, data) -> str:
    if fmt:
        return fmt.lower()
    if out is not None:
        suffix = out.suffix.lower()
        if suffix == ".csv":
            return "csv"
        if suffix == ".json":
            return "json"
    # default: nested dict isn't table-able → json; everything else → table
    if isinstance(data, dict):
        return "json"
    return "table"


def to_json(data) -> str:
    return json.dumps(data, ensure_ascii=False, indent=2, default=str)


def _as_rows(data) -> list[dict]:
    if isinstance(data, dict):
        return [data]
    if isinstance(data, list):
        rows = []
        for item in data:
            if isinstance(item, dict):
                rows.append(item)
            else:
                rows.append({"value": item})
        return rows
    return [{"value": data}]


def _cell(value) -> str:
    if isinstance(value, (dict, list)):
        return json.dumps(value, ensure_ascii=False, default=str)
    return "" if value is None else str(value)


def _columns(rows: list[dict]) -> list[str]:
    cols: list[str] = []
    for row in rows:
        for key in row:
            if key not in cols:
                cols.append(key)
    return cols


def to_csv(data) -> str:
    rows = _as_rows(data)
    cols = _columns(rows)
    buf = io.StringIO()
    writer = csv.DictWriter(buf, fieldnames=cols, extrasaction="ignore")
    writer.writeheader()
    for row in rows:
        writer.writerow({c: _cell(row.get(c)) for c in cols})
    return buf.getvalue()


def decode_disclosure_bodies(detail: dict) -> dict:
    if not isinstance(detail, dict):
        return detail
    messages = detail.get("htmlMessages")
    if isinstance(messages, list):
        for msg in messages:
            if not isinstance(msg, dict):
                continue
            for lang in ("tr", "en"):
                encoded = msg.get(lang)
                if isinstance(encoded, str) and encoded:
                    try:
                        raw = base64.b64decode(encoded).decode("iso-8859-9", errors="replace")
                        msg[f"{lang}_text"] = _strip_html(raw)
                    except (ValueError, TypeError):
                        pass
    return detail


def _strip_html(html: str) -> str:
    text = _STYLE_SCRIPT.sub(" ", html)
    text = _TAGS.sub(" ", text)
    return _WS.sub(" ", text).strip()


def _write_text(text: str, out: Path | None) -> None:
    if out is not None:
        out.write_text(text, encoding="utf-8")
    else:
        print(text)


def _print_table(rows: list[dict], out: Path | None) -> None:
    if out is not None:
        # tables aren't meant for files; fall back to JSON-on-disk
        _write_text(to_json(rows), out)
        return
    if not rows:
        _console.print("[dim](no rows)[/dim]")
        return
    cols = _columns(rows)
    table = Table(show_lines=False)
    for col in cols:
        table.add_column(col, overflow="fold")
    for row in rows:
        table.add_row(*[_cell(row.get(c)) for c in cols])
    _console.print(table)


def output(data, fmt: str | None = None, out: Path | None = None) -> None:
    chosen = resolve_format(fmt, out, data)
    if chosen == "json":
        _write_text(to_json(data), out)
    elif chosen == "csv":
        _write_text(to_csv(data), out)
    else:  # table
        if _is_scalar(data):
            _write_text(str(data), out)
        elif isinstance(data, dict):
            _write_text(to_json(data), out)  # nested → json even if table requested
        else:
            _print_table(_as_rows(data), out)
```

```python
from dataclasses import dataclass
from pathlib import Path
from typing import Optional

import typer
from rich.console import Console

from kap.client import KapClient
from kap.config import load_config
from kap.errors import KapError
from kap.render import decode_disclosure_bodies, output

    add_completion=False,
    no_args_is_help=True,
    help="CLI for the KAP Veri Yayın Servisi REST API (MKK / Borsa İstanbul).",
)
_err_console = Console(stderr=True)


@dataclass
class State:
    env: Optional[str] = None
    fmt: Optional[str] = None
    out: Optional[Path] = None
    verbose: bool = False


@app.callback()
def main(
    ctx: typer.Context,
    env: Optional[str] = typer.Option(None, "--env", help="test | production (overrides .env)."),
    fmt: Optional[str] = typer.Option(None, "--format", "-f", help="table | json | csv."),
    out: Optional[Path] = typer.Option(None, "--out", "-o", help="Write output to this file."),
    verbose: bool = typer.Option(False, "--verbose", "-v", help="Show full error detail."),
):
    ctx.obj = State(env=env, fmt=fmt, out=out, verbose=verbose)


def _make_client(ctx: typer.Context) -> KapClient:
    return KapClient(load_config(ctx.obj.env))


def _run(ctx: typer.Context, fn, *, decode: bool = False, raw_bytes: bool = False, default_name: str | None = None):
    """Execute a client call, render output, and translate errors to clean exits."""
    state: State = ctx.obj
    try:
        client = _make_client(ctx)
        try:
            data = fn(client)
        finally:
            client.close()
    except KapError as exc:
        _err_console.print(f"[red]Error:[/red] {exc}")
        raise typer.Exit(code=1)

    if raw_bytes:
        if data is None:
            _err_console.print("[yellow]Not found.[/yellow]")
            raise typer.Exit(code=1)
        target = state.out or Path(default_name or "attachment.pdf")
        target.write_bytes(data)
        print(f"Saved {len(data)} bytes to {target}")
        return

    if decode and isinstance(data, dict):
        data = decode_disclosure_bodies(data)
    output(data, fmt=state.fmt, out=state.out)
```
